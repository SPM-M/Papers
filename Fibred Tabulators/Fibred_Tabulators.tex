\begin{abstract}
	
	This paper defines tabulators in strict double fibration (\ie, double fibrations of double categories with tabulators). Additionally, we give a Beck-Chevalley Condition for fibred tabulators. Moreover, we show that  fibred tabulators are characterized by a preservation property.
\end{abstract}

\section{Fibrations}

In this section we recall some basic facts about fibrations and their examples.

\begin{defn}
	Let $\fib$ be a functor. A morphism $\mor{f}{X}{Y}$ in $\cat{E}$ is \emph{cartesian} over $\mor{u}{I}{J}$ \ie, $P(f) = u$ if for any $\mor{v}{K}{I}$ in $\cat{B}$ and $\mor{h}{Z}{Y}$ over $u \comp v$ there is a unique morphism $\mor{g}{Z}{X}$ over $v$ with $h = f \comp g$.
	\begin{equation*}
		\xymatrix{
			\cat{E} \ar[d]_{P} && Z \ar@{-->}[r]_{g} \ar@/^1.3pc/[rr]^{h} & X \ar[r]_{f} & Y\\
			\cat{B} && K \ar[r]_{v} & I \ar[r]_{u} & J
		}
	\end{equation*}
	
	A morphism $\mor{k}{X}{Y}$ is called \emph{vertical} if it lies over an identity in $\cat{B}$. For an object $J \in \cat{B}$, the fibre of $P$ at $J$ denoted by $P(J)$ is the subcategory of $\cat{E}$ whose morphisms $k$ lie over $id_J$.
\end{defn}

\begin{defn}
	A functor $\fib$ is called \emph{fibred category} or a \emph{fibration} if for every $Y \in \cat{E}$ and $\mor{u}{I}{PY}$ there is a cartesian morphism $\cart{u}{E}$ over $u$. If $P$ is a fibration, a \emph{cleavage} for $P$ denoted $\varphi(u,E)$ is a choice of cartesian lifts for $u$ at $E$. Whence $P$ is called a cloven fibration.
\end{defn}

\subsection{Examples}

Let $\cat{C}^2$ denote the slice category of $\cat{C}$ over itself
\begin{example}[Fundamental Fibration]
	Let $\cat{C}$ be a category with pullbacks. Then the functor $\mor{\partial_1}{\cat{C}^2}{\cat{C}}$ is a fibration; a commutative square is a cartesian arrow in $\cat{C}^2$ precisely when it is a pullback.
\end{example}

\begin{example}
	The domain functor $\mor{\partial_0}{\slice{\cat{C}}}{\cat{C}}$ is a fibration.
\end{example}

\begin{example}[Family fibration]
	Every category $\cat{C}$ gives rise to a \emph{family fibration} $\mor{\cat{F}}{Fam(\cat{C})}{Set}$. A morphism $(u,\{\varphi_i\}_{i \in I})$ is cartesian when every $\varphi_i$ is an isomorphism. Then $\cat{F}$ is a fibration since given $\mor{u}{I}{J}$ and $\{Y_j\}_{j \in J}$, $\mor{(u,\{1_{Y_{u(i)}}\})}{\{Y_{u(i)}\}_{i \in I}}{\{Y_j\}_{j \in J}}$ is cartesian over $u$. 
\end{example}

\subsection{Fibred Finite Limits}

In \cite{TS} fibred finite limits are given in terms of terminal objects and pullbacks. Furthermore, \cite{JacobsCLTT} captures the notion of fibred terminal objects in $\mathbf{Fib}$ using fibred adjunctions. 

\begin{defn}
	A fibration $\fib$ has fibred terminal objects if every fibre $P(I)$ has a terminal object $\ter{I}$ and they are preserved by cartesianness.
\end{defn}

In $\slicec{\cat{B}}$, the category of fibrations over $\cat{B}$, a fibration $P$ has fibred terminals if the fibred functor $\mor{F}{P}{id_B}$ has a right adjoint. Moreover, this property is characterized by preservation as follows.

\begin{lemma}\label{fibredterminals}
	Let $\cat{B}$ be a category with a terminal object $\ter{B}$. Then $\fib$ has fibred terminals if and only if $\cat{E}$ has a terminal $1_\cat{E}$ and $P(\ter{E})$ is a terminal in $\cat{B}$.
\end{lemma}

\begin{proof}
	Consider the following diagram in $\Fib$
	
	\begin{equation*}
		\xymatrix@=1cm{
			\cat{E} \ar[dd]_{P} \ar@<1.4ex>[rr]^{!_\cat{E}} & \perp & 1 \ar@<1.4ex>[ll]^{\ter{E}} \ar@{=}[dd]^{id}\\
			\\
			B \ar@<1.4ex>[rr]^{!_\cat{B}} & \perp & 1 \ar@<1.4ex>[ll]^{\ter{B}}
		}
	\end{equation*}
\end{proof}

Futhermore, we have the following theorem

\begin{thm}
	If $\cat{B}$ has a terminal object $\ter{B}$ and $\fib$ is a fibration, the following are equivalent
	\begin{enumerate}
		\item $P$ has a terminal object in $\slicec{\cat{B}}$. \\
		\item  $P$ has a terminal object in $\Fib$.
	\end{enumerate}
\end{thm}

\begin{lemma}[\cite{TS}] For a fibration $\fib$ we have that
	
	\begin{enumerate}
		\item A commutative square
		\begin{equation*}
			\xymatrix@=1cm{
				I \ar[d]_{f} \ar[r]^{\varphi_1}_{\text{cart}} & J \ar[d]^{g} \\
				K \ar[r]_{\varphi_2}^{\text{cart}} & L
			}
		\end{equation*}
		in $\cat{E}$ with $\varphi_i$ cartesian and $f$ and $g$ vertical is always a pullback in $\cat{E}$
		
		\item A commutative square of cartesian morphisms in $\cat{E}$ over a pullback in $\cat{B}$ is a pullback in $\cat{E}$.
	\end{enumerate}
	 
\end{lemma}

\begin{defn}
	A fibration $\fib$ has fibred pullbacks if every fibre has pullbacks and they are preserved by cartesianness.
\end{defn}

\begin{lemma}
	If $\fib$ has fibred pullbacks. A pullback in some fibre $P(I)$ is a pullback in $\cat{E}$.
\end{lemma}
Similarly, fibred pullbacks are characterised as follows.
\begin{thm}\label{fibredpb}
	Let $\cat{B}$ be a category with pullbacks. A fibration $\fib$ has fibred pullbacks if and only if $\cat{E}$ has and $P$ preserves pullbacks. 
\end{thm}

The following result is a consequence of Lemma \ref{fibredterminals} and Theorem \ref{fibredpb} 

\begin{thm}
	For a category $\cat{B}$ with finite limits, a fibration $\fib$ has fibred finite limits if and only if $\cat{E}$ has finite limits and $P$ preserves finite limits. 
\end{thm}


\section{Double Categories}

\subsection{Basic Notions}
%In this section investigate fibred limits of double fibrations. In particular, we introduce the notion of \emph{fibred tabulators} \ie, a strict double fibration of double categories with tabulators.

\begin{defn}
	A double category $\dcat{D}$ is a pseudo category in $\Cat$, it consists of the following diagram 
	\begin{equation*}
		\xymatrix{
			\dcat{D}_2 \ar[r]^(.5){\odot} & \dcat{D}_1 \ar@<1ex>[r]^{s} \ar@<-1ex>[r]_{t} & \dcat{D}_0 \ar[l]|-{i}
		}
	\end{equation*}
	We call $\dcat{D}_0$ the category of objects and vertical arrows and $\dcat{D}_1$ the category of proarrows and cells:
	\begin{equation*}
		\xymatrix{
				A \ar[r]|-{\bullet}^-M \ar[d]_f \ar@{}[dr]|{\theta} & B \ar[d]^g \\  
				 C \ar[r]|-{\bullet}_-N & D  
		}
	\end{equation*}
	We refer to $M$ and $N$ as the internal domain and codomain of $\theta$ respectively, whilst $f$ and $g$ are the external source and target respectively. Cells with proarrow (resp. vertical) identities will be called \emph{horizontal} (resp. \emph{vertical}) \emph{globular} cells. Composition is strict in the vertical direction and weaker in the horizontal direction. For convention, we assume our double categories to be pseudo in the sense of \cite{Crutt}.
\end{defn}

\begin{defn}
	Given a double category $\dcat{D}$, we  can form a \emph{horizontal bicategory} $\mathcal{H}(\dcat{D})$, whose objects are the objects of $\dcat{D}$, 1-cells are the proarrows of $\dcat{D}$, and $2$-cells are given by vertical globular cells. Dually, we can form a \emph{vertical 2-category} $\mathcal{V}(\dcat{D})$ whose objects are the objects of $\dcat{D}$, $1$-cells are the vertical arrows, and $2$-cells given by horizontal globular cells. 
\end{defn}


\begin{notation}
	For cells $\alpha$ and $\beta$, we shall write $\frac{\alpha}{\beta}$ as their vertical composition.
\end{notation}

\subsection{Examples}

\begin{example}
	Let $\dcat{D}^2$ be the slice double category of $\dcat{D}$ over itself with external source and target operations coming from $\dcat{D}$.
\end{example}


\begin{example}
	Let $\cat{C}$ be a category with pullbacks, we define a double category $\dcat{S}\textbf{pan}(\cat{C})$ whose vertical category is $\cat{C}$, whose proarrows $\pro{A}{B}{}$  are spans $\hspan{A}{C}{B}{f}{g}$ in $\cat{C}$, and whose cells are commuting squares:
	\begin{equation*}
		\xymatrix{
			A \ar[d] & C \ar[d] \ar[l] \ar[r] & B \ar[d] \\
			D & F \ar[l] \ar[r] & E
		}
	\end{equation*}
	in $\cat{C}$. Horizontal composition is given by pullbacks.
\end{example}

\begin{example}
	The double category $\dcat{R}$el(Set) has objects as sets and vertical arrows as set-functions. The proarrows are given by relations and there is a unique cell
	\begin{equation*}
		\xymatrix{
			A  \ar[d]_f \ar[r]|-{\bullet}^R \ar@{}[dr]|{\Downarrow} & B \ar[d]^g \\
			C \ar[r]|-{\bullet}_{S}& D
		}
	\end{equation*}
	whenever $R(a,b) \implies S(f(a),g(b))$ holds for all $a \in A, b \in B$.
\end{example}


\begin{defn}
	Let $\dcat{E}$ and $\dcat{B}$ be pseudo categories internal to a $2$-category $\cat{K}$ with $2$-pullbacks. A \emph{lax functor} $\dfib{L}{E}{B}$ consists of arrows $\funct{L_0}{E_0}{D_0}$ and $\funct{L_1}{E_1}{B_1}$ together with comparison morphisms given by 

	\begin{equation*}
		\vcenter{\vbox{\xymatrix{
				E_2 \ar[r]^(.5){\odot_E}  \ar[d]_{L_2} \ar@{}[dr]|{\xRightarrow{\phi}}& E_1 \ar[d]^{L_1}\\
				B_2 \ar[r]_(.5){\odot_B} & B_1
		}}}
		\qquad \qquad \qquad
		\vcenter{\vbox{\xymatrix{
				E_0 \ar[d]_{L_0} \ar[r]^{i} \ar@{}[dr]|{\xRightarrow{\iota}} & E_1 \ar[d]^{L_1}\\
				B_0 \ar[r]_{i} & B_1	
		}}}
	\end{equation*}
	such that the source and target are strictly preserving. Moreover, $\phi$, $\iota$, and the associators and unitors of $\dcat{E}$ and $\dcat{B}$  satisfy the usual coherence conditions as in \cite{NMF}. A lax functor is called $\mathbf{strict}$ if $\phi$ and $\iota$ are identities.
\end{defn}	

\begin{remark}
	
	A strict functor in $\Cat$ corresponds to a strict functor between double categories.
\end{remark}

We recall the notion of double limits \ie, limits in double categories introduced in \cite{LMP}. We define tabulators, which will be crucial in the next section. In particular, tabulators of proarrow identities, which are related to cotensors in the case where our double category $\dcat{E}$ is reduced to a $2$-category, see \cite{RS} for more details. 


\begin{defn}[\cite{SpanDPP}] \label{vp.bs}
	A double category $\dcat{E}$ has \emph{vertical pullbacks} if the diagram
	\begin{equation*}
		\xymatrix{
			\dcat{E}_2 \ar[r]^(.5){\odot} & \dcat{E}_1 \ar@<1ex>[r]^{s} \ar@<-1ex>[r]_{t} & \dcat{E}_0 \ar[l]|-{i}
		}
	\end{equation*}
	lies in $\mathbf{PBCat}$ the category of categories with pullbacks and pullback-preserving functors.
\end{defn}

A functor $\funct{G}{\dcat{E}}{\dcat{A}}$ between double categories with vertical pullbacks preserves pullbacks if it preserves pullbacks componentwise. When this is the case, we shall call $G$ a pullback-preserving functor. For the rest of the paper, we shall refer to vertical pullbacks by v-pullbacks.

\begin{remark}
	The universal property of v-pullbacks in double categories generalizes the universal property of 2-pullbacks. $\odot$ preserving pullbacks amounts to a  functorial choice of pullbacks, see \cite{LMP}.
\end{remark}
	
\begin{defn}
	The  $1$-\emph{tabulator} of a proarrow $\pro{A}{B}{M}$ consists of an object $TM$ together a cell
	\begin{equation*}
		\xymatrix{
			TM \ar@{=}[r] \ar[d]_{\pi_A} \ar@{}[dr]|{\pi} & TM \ar[d]^{\pi_B} \\
			A \ar[r]|-{\bullet}_{M} & B
		}
	\end{equation*}
	such that for any other cell $\xi$ into $M$ there exists a unique vertical arrow $\funct{r}{R}{TM}$ with $ \xi = \frac{1_{r}}{\pi}$ as depicted below.
	\begin{equation*}
		\vcenter{\vbox{\xymatrix{
					R \ar@{=}[r] \ar[d]_{\xi_A} \ar@{}[dr]|{\xi} & R \ar[d]^{\xi_B} \\
					A \ar[r]|-{\bullet}_{M} & B
		}}} \quad = 
		\quad
		\vcenter{\vbox{\xymatrix{
					R \ar@{=}[r] \ar@{-->}[d]_{r} \ar@{}[dr]|{1_r} & R \ar@{-->}[d]^{r} \\
					TM \ar@{=}[r] \ar[d]_{\pi_A} \ar@{}[dr]|{\pi} & TM \ar[d]^{\pi_B} \\
					A \ar[r]|-{\bullet}_{M} & B
		}}}
	\end{equation*}
	This is known as the 1-dimensional universal property of tabulators. 
\end{defn}


\begin{example}
	In $\dspan{C}$ the tabulator of a span $\hspan{A}{X}{B}{f}{g}$ is given by $X$ together with the cell
	\begin{equation*}
		\xymatrix{
			X \ar[d]_{f} & X \ar[l]_{id_X} \ar[r]^{id_X} \ar[d]^{id_X} & X \ar[d]^{g} \\
			A & X \ar[l]^{f} \ar[r]_{g} & B
		}
	\end{equation*}
\end{example}

\begin{example}
	In $\mathbb{R}\text{el}(\text{Set})$, for a relation $\funct{\cat{R}}{A \times B}{\mathsf{2}}$, the tabulator of $\cat{R}$ is computed by pullback
	\begin{equation*}
		\xymatrix{
			T\cat{R} \ar[r] \ar@{}[d]^(0.2){}="start" \ar@{>->} "start";[d]  \pullbackcorner & \mathsf{1}  \ar[d]^{\top} \\
			A \times B \ar[r]_{\cat{R}} & \mathsf{2}
		}
	\end{equation*}
	where $\mathsf{2} = \{\bot \to \top\}$.
\end{example}
Although there is a 2-dimensional property of tabulators, it will play no role here. Hence, we shall refer to 1-tabulators as tabulators. We have the following result 

\begin{lemma}[\cite{LMP}]
	A double category $\dcat{D}$ has tabulators if and only if  the functor $\funct{i}{\cat{D}_0}{\cat{D}_1}$ has a right adjoint.
\end{lemma}

\begin{thm}
	The double category $\dcat{D}$ has all small (resp. finite) double limits if and only if it has small (resp. finite) double products, double equalizers and tabulators.
\end{thm}

\section{Double Fibrations}
\begin{lemma}[\cite{Crutt}]
	Given a diagram $s \colon P_1 \to P_0 \leftarrow P_1 \colon t$ in $\mathbf{cFib}$, a pseudo category structure in $\mathbf{Fib}$ (resp. $\mathbf{cFib}$) is the same as two pseudo double category structure obtained by applying the top and bottom 2-functors,
\end{lemma}

\begin{defn}[\cite{Crutt}]
	A strict double functor $\dfib{P}{E}{B}$ between pseudo double categories given by 
	
	\begin{equation*}
		\xymatrix@=0.5cm{
			\cat{E}_2 \ar[rr]^{\odot_\cat{E}} \ar[dd]_{\cat{P}_2} && \cat{E}_1 \ar[dd]_{\cat{P}_1} \ar@<1ex>[rr]^{s_\cat{E}} \ar@<-1ex>[rr]_{t_\cat{E}} && \cat{E}_0 \ar[dd]^{\cat{P}_0} \ar[ll]|-{i_\cat{E}} \\
			\\
			\cat{B}_2 \ar[rr]_{\odot_\cat{B}} && \cat{B}_1 \ar@<1ex>[rr]^{s_\cat{B}} \ar@<-1ex>[rr]_{t_\cat{B}} && \cat{B}_0 \ar[ll]|-{i_\cat{B}}
		}
	\end{equation*}
	defines a \textbf{double fibration} when $P_0$ and $P_1$ are cloven fibrations, such that $s_{\cat{E}}$ and $t_{\cat{B}}$ preserve cleavages simultaneously, and $\odot_\cat{E}$ and $i_\cat{E}$ preserve cartesian morphisms.
\end{defn}

\begin{remark}
	A  double fibration $\dfib{P}{E}{B}$ is a pseudo category internal to $\mathbf{Fib}$ such that the source and target maps preserve chosen cleavages. A cleavage for $P$ is chosen componentwise. Since $s_{\cat{E}}$ and $t_{\cat{B}}$ preserve cleavages, they are morphisms in $\mathbf{cFib}$.
\end{remark}

\begin{example}(Domain double fibration)
	Let $\dcat{D}$ be a double category. The functor $\funct{\textbf{dom}}{\dcat{D}^2}{\dcat{D}}$ sending an arrow to its domain and a cell to its internal domain is a double fibration. 
\end{example}

\begin{example}(Codomain double fibration)
	Let $\dcat{D}$ be a pseudo category in the 2-category of finitely-complete categories, limit-preserving functors and natural transformations. Suppose the source and target strictly preserve limits, and the identity and  composition preserve limits up to isomorphism. Then $\funct{\textbf{cod}}{\dcat{D}^2}{\dcat{D}}$ is a double fibration. Cartesian morphisms of $\dcat{D}_1$ are given by pullbacks of cells.
\end{example}

\begin{example}[\cite{ElenaMesitiPronk}]
	The image double functor $\funct{\text{Im}}{\cspan}{\mathbb{R}\text{el}}$ with $\funct{\text{Im}}{Set}{Set}$ and $\funct{\text{Im}}{\cspan_1}{\mathbb{R}\text{el}_1}$ sending a span $\hspan{A}{C}{B}{f}{g}$ to  the image of $C \xlongrightarrow{(f,g)} A \times B$ seen as a relation and a cell
	\begin{equation*}
		\xymatrix{
			A \ar[d]_u & C \ar[d]^s \ar[l]_f \ar[r]^g & B \ar[d]^v \\
			D & F \ar[l]^l \ar[r]_k & E
		}
	\end{equation*}
	to 
	\begin{equation*}
		\xymatrix{
			\text{Im}(f,g) \ar[rr]^{\tilde{s}} \ar@{}[d]^(0.2){}="start" \ar@{^{(}->} "start";[d]  && \text{Im}(l,k) \ar@{}[d]^(0.2){}="start" \ar@{^{(}->} "start";[d] \\
			A\times B \ar[rr]_{u \times v} && D \times E
		}
	\end{equation*}
	where $\funct{\tilde{s}}{\text{Im}(f,g)}{\text{Im}(l,k)}$ is the function mapping $(f(c),g(c))$ to $(l(s(c)), k(s(c)))$.
\end{example}


\subsection{Fibred double limits}
For the rest of this section, we fix $\dcat{B}$ a pseudo category internal in $\mathbf{PBCat}$ as in Definition \ref{vp.bs}. 

\begin{defn}
	A double fibration $\dfib{P}{E}{B}$ has fibred terminal objects if every fibre of $\cat{P}_0$ and $\cat{P}_1$ has a terminal object and these are preserved by cartesianness respectively. 
\end{defn}

$\cat{P}_0$ is ordinary fibration with fibred terminals. Let $M \in \dcat{D}_1$ be an object, the fibre $\cat{P}_1(M)$ over $M$ consists of proarrows $A$ such that $\cat{P}_1{A} = M$ and cells 
\begin{equation*}
	\xymatrix{
		{} \ar[r]|-{\bullet}^{A} \ar[d]_{f} \ar@{}[dr]|{\alpha} & {} \ar[d]^{g} \\
		{} \ar[r]|-{\bullet}_{B} & {}
	}
\end{equation*}

over the globular cell

\begin{equation*}
	\xymatrix{
		{} \ar[r]|-{\bullet}^M \ar@{=}[d] \ar@{}[dr]|{\ipro{M}} & {} \ar@{=}[d] \\
		{} \ar[r]|-{\bullet}_M & {}
	}
\end{equation*}
We shall refer to such cells as \emph{vertical cells}. $\compdfib{P}{E}{B}{1}$ has fibred terminals in the sense that for every $M \in \dcat{B}_1$ there is an object $1^{\bullet}_T \in \cat{P}_1(M)$ such that for every cell $\funct{\psi}{N}{M}$ and an object $X \in \cat{P}_1(N)$ there is a unique cell
\begin{equation*}
	\xymatrix{
		A \ar[r]|-{\bullet}^{X} \ar[d]_{l} \ar@{}[dr]|{\tau} & B \ar[d]^{k} \\
		T \ar[r]|-{\bullet}_{1^{\bullet}_T} & T
	}
\end{equation*}
over $\psi$.

\begin{lemma}
	Let $\dcat{B}$ have terminal objects. A double fibration $\dfib{P}{E}{B}$ has fibred terminals if and only if $\dcat{E}$ has terminals and $P_0$ and $P_1$ preserve terminal objects.
\end{lemma}

%In a double fibration $\dfib{P}{E}{B}$ the functor $\funct{i}{\cat{E}}{\cat{E}_1}$
\begin{prop}\label{vc}
	Let $\dfib{P}{E}{B}$ be a double fibration. A commuting cube 
	\begin{equation*}
		\xymatrix{
			&K \ar@{=}[rr] \ar[dd]^(.3){v}|\hole \ar[dl]_{q} && K \ar[dd]^{v} \ar[dl]^{q} \\
			L \ar@{=}[rr] \ar[dd]_{p} && L \ar[dd]^(.3){p} \\
			& I \ar@{=}[rr]|\hole \ar[dl]_{u}  && I \ar[dl]^{u}  \\
			J \ar@{=}[rr] && J
		}
	\end{equation*}
	where the top and bottom squares are cartesian and the front and back squares are vertical is always a vertical pullback in $\dcat{E}_1$.
\end{prop}

\begin{proof}
	The top and bottom faces are cartesian since the functor $\funct{i}{\cat{E}_0}{\cat{E}_1}$ preserves cartesian morphisms. Consider the a cone over the front and back faces \ie, for an object $1^{\bullet}_X$ we have cells
	\begin{equation*}
			\vcenter{\vbox{\xymatrix{
						X \ar[d]_{x_I} \ar@{=}[r] \ar@{}[dr]|{\gamma_I} & X \ar[d]^{x_I} \\					
						I \ar@{=}[r] & I
			}}} \qquad
			\vcenter{\vbox{\xymatrix{
						X \ar[d]_{x_L} \ar@{=}[r] \ar@{}[dr]|{\gamma_L} & X \ar[d]^{x_L} \\					
						L \ar@{=}[r] & L
			}}}
		\end{equation*}
		such that
		\begin{equation*}
			\vcenter{\vbox{\xymatrix{
						X \ar[d]_{x_I} \ar@{=}[r] \ar@{}[dr]|{\gamma_I} & X \ar[d]^{x_I}	 \\
						I \ar[d]_{u} \ar@{=}[r] \ar@{}[dr]|{1_u} & I \ar[d]^{u} \\
						J \ar@{=}[r] & J
			}}} \quad
			=
			\quad
			\vcenter{\vbox{\xymatrix{
						X \ar[d]_{x_L} \ar@{=}[r] \ar@{}[dr]|{\gamma_L} & X \ar[d]^{x_L}	 \\
						L \ar[d]_{p} \ar@{=}[r] \ar@{}[dr]|{1_p} & L \ar[d]^{p} \\
						J \ar@{=}[r] & J
			}}}
	\end{equation*}
	
	and a cell 
	
	\begin{equation*}
		\xymatrix{
			A \ar[r]|-{\bullet}^{N} \ar[d]_{l} \ar@{}[dr]|{\psi} & B \ar[d]^{k} \\
			C \ar[r]|-{\bullet}_{M} & D
		}
	\end{equation*}
	in $\cat{B}_1$. Since the bottom square is cartesian there is a unique cell 
	
	\begin{equation*}
		\xymatrix@=1cm{
			X \ar@{-->}[d]_{x_I} \ar@{=}[r] \ar@{}[dr]|{\tilde{\gamma}_I} & X \ar@{-->}[d]^{x_I} \\
			I \ar@{=}[r] & I
		}
	\end{equation*}
	Since the top and bottom squares lie over $\psi$, there is another unique cell $\tilde{\gamma}_K$ since the top square is cartesian. But $\tilde{\gamma}_I$ is given by the composition
	\begin{equation*}
		\vcenter{\vbox{\xymatrix@=1cm{
					X \ar[d]_{x_K} \ar@{=}[r] \ar@{}[dr]|{\tilde{\gamma}_K} & X \ar[d]^{x_K} \\
					K \ar[d]_{v} \ar@{=}[r] \ar@{}[dr]|{1_v} & K \ar[d]^{v} \\
					I \ar@{=}[r] & I 
		}}} \qquad
		= 
		\qquad
		\vcenter{\vbox{\xymatrix{
				X \ar[d]_{x_I} \ar@{=}[r] \ar@{}[dr]|{\tilde{\gamma}_I} & X \ar[d]^{x_I} \\
				I \ar@{=}[r] & I	
		}}}
	\end{equation*}
	Thus $\tilde{\gamma}_K$ is the unique cone morphism.
	
\end{proof}

\begin{thm}
	If $\dfib{P}{E}{B}$ is a double fibration with  fibred pullbacks, then every pullback in some fibre is  a pullback in $\dcat{E}$.
\end{thm}
\begin{proof}
	Clearly, this is true for $\cat{P}_0$. To show this for $\cat{P}_1$ let 
	\begin{equation*}\label{p.bcube} \tag{$\clubsuit$}
		\xymatrix{
			&L \ar[rr]|{\bullet}^{\mathbi{a}} \ar[dd]_(.7){v}|\hole \ar[dl]_{q} && L^\prime \ar[dd]^{v^\prime} \ar[dl]^{q^\prime} \\
			K \ar[rr]|(.6){\bullet}^(.6){\mathbi{b}} \ar[dd]_{p} && K^\prime \ar[dd]^(.3){p^\prime} \\
			& I \ar[rr]|(.3){\bullet}_(.3){\mathbi{c}}|\hole \ar[dl]_{u}  && I^\prime \ar[dl]^{u^\prime}  \\
			J \ar[rr]|{\bullet}_{\mathbi{d}} && J^\prime
		}
	\end{equation*}
	be a pullback in $\cat{E}_1$ over $\pro{A}{B}{M} \in \cat{B}_1$ and 
	\begin{equation*}
		\xymatrix{
				A \ar[r]|-{\bullet}^{N} \ar[d]_{f} \ar@{}[dr]|{\psi} & B \ar[d]^{g} \\
				C \ar[r]|-{\bullet}_{M} & D
		}
	\end{equation*}
	be a cell in $\cat{B}_1$. Consider the following diagram
	\begin{equation*}

			C \ar@/_2.5pc/[ddd]_{c_K} \ar[d]_{c_I} \ar@{=}[r] \ar@{}[dr]|{\alpha} & C \ar[d]^{c_{I^\prime}} \ar@/^2.5pc/[ddd]^{c_{K^\prime}} \\

	with $p \comp c_K = u \comp c_I$ and $p^\prime \comp c_{K^\prime} = u^\prime \comp c_{I^\prime}$ for an object $1^{\bullet}_C \in \cat{E}_1$ \ie, a double cone over $\eta$ and $\gamma$. The cells
	\begin{equation*}

		}}}
	\end{equation*}
	lie over the same cell $\psi$. Futhermore, let 
	\begin{equation*}

					C \ar[d]_{c_K} \ar@{=}[r] \ar@{}[dr]|{\chi} & C \ar[d]^{c_{K^\prime}} \\

		}}}
		\qquad
		\vcenter{\vbox{\xymatrix{
					C \ar[d]_{c_K} \ar@{=}[r] \ar@{}[dr]|{\chi} & C \ar[d]^{c_{K^\prime}} \\
					f^*I \ar[r]|{\bullet}^{\mathbi{c}^*} \ar[d]_{\cleav{f}{I}} \ar@{}[dr]|{\text{cart}} & g^*I^\prime \ar[d]^{\cleav{g}{I^\prime}} \\
					I \ar[r]|{\bullet}_{\mathbi{c}} & I^\prime
		}}}
	\end{equation*}
	be the vertical-cartesian factorizations of $\chi$ and $\alpha$ respectively. The image of $\clubsuit$ under change of base is also a pullback, there is a vertical cell 
	\begin{equation*}
		\xymatrix{
			C \ar[d]_{\bar{c}} \ar@{=}[r] \ar@{}[dr]|{\mu} & C \ar[d]^{\bar{c'}} \\

		}
	\end{equation*}
	with 
	\begin{equation*}

		}}} \quad ,\
		\quad
		\vcenter{\vbox{\xymatrix{

					f^*I \ar[r]|{\bullet}_{\mathbi{c}^*} & g^*I^\prime

			& J \ar@{=}[rr] && J \ar[rr]|{\bullet}_{d} && J' \ar@{=}[rr] && J'

	\end{equation*}
	with the correspondence $i = I,\ j = K'$ and $i = K,\ j = I'$ and the cells
	\begin{equation*}

		}}}\quad
		,\
		\quad

\end{proof}

\begin{defn}
	A double fibration $\dfib{P}{E}{B}$ has fibred tabulators if every fibre of $\cat{P}_1(M)$ has tabulators.
\end{defn}

\begin{prop}
	For a double fibration $\dfib{P}{E}{B}$ with fibred tabulators, cartesian lifts of proarrow identities $1^{\bullet}_X$ can always be reduced to tabulators of proarrow identities.
\end{prop}

\begin{lemma}
	If a double fibration $\dfib{P}{E}{B}$ has fibred tabulators, then a tabulator in some fibre is a tabulator in $\dcat{E}$.
\end{lemma}

\begin{proof}
	Let 
	\begin{equation*}
		\xymatrix{
			E \ar@{=}[r]\ar[d]_{f} \ar@{}[dr]|{1_f} & E \ar[d]^{f} \\
			D \ar@{=}[r] & D
		}
	\end{equation*}
	be a cell in $\dcat{B}$ and 
	\begin{equation*}
		\xymatrix@=1cm{
			Tu \ar@{=}[r] \ar[d]_{\pi_A} \ar@{}[dr]|{\pi} & Tu \ar[d]^{\pi_B} \\
			{A} \ar[r]|{\bullet}_u & {B}
		}
	\end{equation*}
	be a tabulator over $M$. Consider a double cone over $\pi$ given by 
	\begin{equation*}
		\xymatrix@=1cm{
			&C \ar[dl]_{p} \ar[ddl]|\hole^(.59){h_1} \ar[dr]^{p} \ar[ddr]|\hole_(.59){h_2}\\
			Tu \ar[d]_{\pi_A} \ar@{=}[rr] \ar@{}[drr]|{\pi} && Tu \ar[d]^{\pi_B} \\
			A \ar[rr]|{\bullet}_u && B
		}
	\end{equation*}
	with $\pi_A \comp p = h_1$ and $\pi_B \comp p = h_2$. Clearly, the cell
	\begin{equation*}
		\xymatrix@=1cm{
			C \ar@{=}[r] \ar[d]_{h_1} \ar@{}[dr]|{\phi} & C \ar[d]^{h_2} \\
			A \ar[r]|{\bullet}_u & B
		}
	\end{equation*}
	lies over $1_f$. Let
	\begin{equation*}
		\vcenter{\vbox{\xymatrix@=1cm{
					C \ar@{=}[r] \ar[d]_{h_1} \ar@{}[dr]|{\phi} & C \ar[d]^{h_2} \\
					A \ar[r]|{\bullet}_u & B
		}}} \quad = \quad 
		\vcenter{\vbox{\xymatrix{
					C \ar@{=}[r] \ar[d]_{c_A} \ar@{}[dr]|{\tau} & C \ar[d]^{c_B} \\
					f^*A \ar[d]_{\cleav{f}{A}} \ar[r]|{\bullet}_{u^*} \ar@{}[dr]|{\text{cart}} & f^*B \ar[d]^{\cleav{f}{B}} \\
					A \ar[r]|{\bullet}_{u} & B
		}}}
	\end{equation*}
	be the vertical-cartesian factorization of $\phi$. Since every fibre has a tabulator, there is a vertical cell
	\begin{equation*}
		\xymatrix{
			C \ar@{=}[r] \ar[d]_{p^*} \ar@{}[dr]|{1_{p^*}} & C \ar[d]^{p^*} \\
			Tu^* \ar@{=}[r] & Tu^*
		}
	\end{equation*}
	with $\frac{1_{k^*}}{\delta} = \tau$ where $\delta$ is the tabulator over $E$. As illustrated in the following diagram
	\begin{equation*}
		\xymatrix{
			&& C \ar[ddrr]^{x_B} \ar[ddll]_{x_A} \ar[d]^{1_{k^*}} \\
			&& Tu^* \ar[dll]^{\pi_A^*} \ar[drr]_{\pi_B^*} \ar[dd]|\hole_(.20){t}
			\\
			f^*A \ar[dd]_{\cleav{f}{A}}\ar[rrrr]|(.6){\bullet}_(.6){u^*} &&&& f^*B \ar[dd]^{\cleav{f}{B}}\\
			&& Tu \ar[dll]_{\pi_A} \ar[drr]^{\pi_B} \\
			A \ar[rrrr]|{\bullet}_{u} &&&& B
		}
	\end{equation*}
	the cell $1_k \colon 1^{\bullet}_{C} \longrightarrow 1^{\bullet}_{Tu}$ given by
	\begin{equation*}
		\vcenter{\vbox{\xymatrix{
					C \ar[d]_{k} \ar@{=}[r] \ar@{}[dr]|{1_k} & C \ar[d]^{k}\\
					Tu \ar@{=}[r] & Tu
		}}}
		\quad
		=
		\quad
		\vcenter{\vbox{\xymatrix{
					C \ar[d]_{k^*} \ar@{=}[r] \ar@{}[dr]|{1_{k^*}} & X \ar[d]^{k^*}\\
					Tu^* \ar[d]_{t} \ar@{}[dr]|{\substack{1_t \\ \text{cart}}} \ar@{=}[r] \ & Tu^* \ar[d]^{t} \\
					Tu \ar@{=}[r] & Tu
		}}}
	\end{equation*}
	where the bottom cell on the right is cartesian over $1_f$, is the desired mediating cell. 
%	If $\alpha ^ \prime = \frac{\omega^\prime}{\beta}$ with $\omega^\prime$ vertical,  it holds that $\omega^\prime = \omega$ as both are mediating cells to the tabulator over $v$ for the cone given by vertical arrows $x_A$ and $x_B$, therefore $\alpha^\prime = \alpha$.
\end{proof}

\begin{remark}
	The above formulation works over horizontal globular cells, in general we would not obtain a tabulator under change of base.
\end{remark}

\begin{remark}
	It is clear from Proposition \ref{vc} that for any commuting cube where the front square is a tabulator and the bottom square is cartesian, then the back square is a tabulator if and only if the cube is a pullback.
\end{remark}
