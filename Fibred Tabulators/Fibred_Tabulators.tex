\begin{abstract}
	
	This paper defines tabulators in strict double fibration (\ie, double fibrations of double categories with tabulators). Additionally, we give a Beck-Chevalley Condition for fibred tabulators. Moreover, we show that  fibred tabulators are characterized by a preservation property.
\end{abstract}

\section{Fibrations}

In this section we recall some basic facts about fibrations and their examples.

\begin{defn}
	Let $\fib$ be a functor. A morphism $\mor{f}{X}{Y}$ in $\cat{E}$ is \emph{cartesian} over $\mor{u}{I}{J}$ \ie, $P(f) = u$ if for any $\mor{v}{K}{I}$ in $\cat{B}$ and $\mor{h}{Z}{Y}$ over $u \comp v$ there is a unique morphism $\mor{g}{Z}{X}$ over $v$ with $h = f \comp g$.
	\begin{equation*}
		\xymatrix{
			\cat{E} \ar[d]_{P} && Z \ar@{-->}[r]_{g} \ar@/^1.3pc/[rr]^{h} & X \ar[r]_{f} & Y\\
			\cat{B} && K \ar[r]_{v} & I \ar[r]_{u} & J
		}
	\end{equation*}
	
	A morphism $\mor{k}{X}{Y}$ is called \emph{vertical} if it lies over an identity in $\cat{B}$. For an object $J \in \cat{B}$, the fibre of $P$ at $J$ denoted by $P(J)$ is the subcategory of $\cat{E}$ whose morphisms $k$ lie over $id_J$.
\end{defn}

\begin{defn}
	A functor $\fib$ is called \emph{fibred category} or a \emph{fibration} if for every $Y \in \cat{E}$ and $\mor{u}{I}{PY}$ there is a cartesian morphism $\cart{u}{E}$ over $u$. If $P$ is a fibration, a \emph{cleavage} for $P$ denoted $\varphi(u,E)$ is a choice of cartesian lifts for $u$ at $E$. Whence $P$ is called a cloven fibration.
\end{defn}

\subsection{Examples}

Let $\cat{C}^2$ denote the slice category of $\cat{C}$ over itself
\begin{example}[Fundamental Fibration]
	Let $\cat{C}$ be a category with pullbacks. Then the functor $\mor{\text{cod}}{\cat{C}^2}{\cat{C}}$ is a fibration; a commutative square is a cartesian arrow in $\cat{C}^2$ precisely when it is a pullback.
\end{example}

\begin{example}
	The domain functor $\mor{\text{dom}}{\slice{\cat{C}}}{\cat{C}}$ is a fibration.
\end{example}

\begin{example}[Family fibration]
	Every category $\cat{C}$ gives rise to a \emph{family fibration} $\mor{\cat{F}}{Fam(\cat{C})}{Set}$. A morphism $(u,\{\varphi_i\}_{i \in I})$ is cartesian when every $\varphi_i$ is an isomorphism. Then $\cat{F}$ is a fibration since given $\mor{u}{I}{J}$ and $\{Y_j\}_{j \in J}$, $\mor{(u,\{1_{Y_{u(i)}}\})}{\{Y_{u(i)}\}_{i \in I}}{\{Y_j\}_{j \in J}}$ is cartesian over $u$. 
\end{example}

\subsection{Fibred Finite Limits}

In \cite{TS} fibred finite limits are given in terms of terminal objects and pullbacks. Furthermore, \cite{JacobsCLTT} captures the notion of fibred terminal objects in $\mathbf{Fib}$ using fibred adjunctions. 

\begin{defn}
	A fibration $\fib$ has fibred terminal objects if every fibre $P(I)$ has a terminal object $\ter{I}$ and they are preserved by cartesianness.
\end{defn}

In $\slicec{\cat{B}}$, the category of fibrations over $\cat{B}$, a fibration $P$ has fibred terminals if the fibred functor $\mor{F}{P}{id_B}$ has a right adjoint. Moreover, this property is characterized by preservation as follows.

\begin{lemma}\label{fibredterminals}
	Let $\cat{B}$ be a category with a terminal object $\ter{B}$. Then $\fib$ has fibred terminals if and only if $\cat{E}$ has a terminal $1_\cat{E}$ and $P(\ter{E})$ is a terminal in $\cat{B}$.
\end{lemma}

\begin{proof}
	Consider the following diagram in $\Fib$
	
	\begin{equation*}
		\xymatrix@=1cm{
			\cat{E} \ar[dd]_{P} \ar@<1.4ex>[rr]^{!_\cat{E}} & \perp & 1 \ar@<1.4ex>[ll]^{\ter{E}} \ar@{=}[dd]^{id}\\
			\\
			B \ar@<1.4ex>[rr]^{!_\cat{B}} & \perp & 1 \ar@<1.4ex>[ll]^{\ter{B}}
		}
	\end{equation*}
\end{proof}

Futhermore, we have the following theorem

\begin{thm}
	If $\cat{B}$ has a terminal object $\ter{B}$ and $\fib$ is a fibration, the following are equivalent
	\begin{enumerate}
		\item $P$ has a terminal object in $\slicec{\cat{B}}$. \\
		\item  $P$ has a terminal object in $\Fib$.
	\end{enumerate}
\end{thm}

\begin{defn}
	A fibration $\fib$ has fibred pullbacks if every fibre has pullbacks and they are preserved by cartesianness.
\end{defn}

Similarly, fibred pullbacks can be characterised as follows.
\begin{thm}\label{fibredpb}
	Let $\cat{B}$ be a category with pullbacks. A fibration $\fib$ has fibred pullbacks if and only if $\cat{E}$ has and $P$ preserves pullbacks. 
\end{thm}

The following result is a consequence of Lemma \ref{fibredterminals} and Theorem \ref{fibredpb} 

\begin{thm}
	For a category $\cat{B}$ with finite limits, a fibration $\fib$ has fibred finite limits if and only if $\cat{E}$ has finite limits and $P$ preserves finite limits. 
\end{thm}

\section{Double Categories}

\subsection{Basic Notions}
%In this section investigate fibred limits of double fibrations. In particular, we introduce the notion of \emph{fibred tabulators} \ie, a strict double fibration of double categories with tabulators.

\begin{defn}
	A double category $\dcat{D}$ is a pseudo category in $\Cat$, it consists of the following diagram 
	\begin{equation*}
		\xymatrix{
			\dcat{D}_1 \times_{\dcat{D}_0}\dcat{D}_1 \ar[r]^(.6){\odot} & \dcat{D}_1 \ar@<1ex>[r]^{s} \ar@<-1ex>[r]_{t} & \dcat{D}_0 \ar[l]|-{i}
		}
	\end{equation*}
	We call $\dcat{D}_0$ the category of objects and vertical arrows and $\dcat{D}_1$ the category of proarrows and cells:
	\begin{equation*}
		\xymatrix{
				A \ar[r]|-{\bullet}^-M \ar[d]_f \ar@{}[dr]|{\theta} & B \ar[d]^g \\  
				 C \ar[r]|-{\bullet}_-N & D  
		}
	\end{equation*}
	We refer to $M$ and $N$ as the internal domain and codomain of $\theta$ respectively, whilst $f$ and $g$ are the external source and target respectively. Cells with proarrow (resp. vertical) identities will be called \emph{horizontal} (resp. \emph{vertical}) \emph{globular} cells. Composition is strict in the vertical direction and weaker in the horizontal direction. For convention, we assume our double categories to be pseudo in the sense of \cite{Crutt}.
\end{defn}

\begin{defn}
	Given a double category $\dcat{D}$, we  can form a \emph{horizontal bicategory} $\mathcal{H}(\dcat{D})$, whose objects are the objects of $\dcat{D}$, 1-cells are the proarrows of $\dcat{D}$, and $2$-cells are given by vertical globular cells. Dually, we can form a \emph{vertical 2-category} $\mathcal{V}(\dcat{D})$ whose objects are the objects of $\dcat{D}$, $1$-cells are the vertical arrows, and $2$-cells given by horizontal globular cells. 
\end{defn}


\begin{notation}
	For cells $\alpha$ and $\beta$, we shall write $\frac{\alpha}{\beta}$ as their vertical composition.
\end{notation}

\subsection{Examples}

\begin{example}
	Let $\dcat{D}^2$ be the slice double category of $\dcat{D}$ over itself with external source and target operations coming from $\dcat{D}$.
\end{example}


\begin{example}
	Let $\cat{C}$ be a category with pullbacks, we define a double category $\dcat{S}\textbf{pan}(\cat{C})$ whose vertical category is $\cat{C}$, whose proarrows $\pro{A}{B}$  are spans $\hspan{A}{C}{B}{f}{g}$ in $\cat{C}$, and whose cells are commuting squares:
	\begin{equation*}
		\xymatrix{
			A \ar[d] & C \ar[d] \ar[l] \ar[r] & B \ar[d] \\
			D & F \ar[l] \ar[r] & E
		}
	\end{equation*}
	in $\cat{C}$. Horizontal composition is given by pullbacks.
\end{example}

\begin{example}
	The double category $\dcat{R}$el(Set) has objects as sets and vertical arrows as set-functions. The proarrows are given by relations and there is a unique cell
	\begin{equation*}
		\xymatrix{
			A  \ar[d]_f \ar[r]|-{\bullet}^R \ar@{}[dr]|{\Downarrow} & B \ar[d]^g \\
			C \ar[r]|-{\bullet}_{S}& D
		}
	\end{equation*}
	whenever $R(a,b) \implies S(f(a),g(b))$ holds for all $a \in A, b \in B$.
\end{example}


\begin{defn}
	Let $\dcat{E}$ and $\dcat{B}$ be pseudo categories internal to a $2$-category $\cat{K}$ with $2$-pullbacks. A \emph{lax functor} $\dfib{L}{E}{B}$ consists of arrows $\funct{L_0}{E_0}{D_0}$ and $\funct{L_1}{E_1}{B_1}$ together with comparison morphisms given by 

	\begin{equation*}
		\vcenter{\vbox{\xymatrix{
				E_1 \times_{E_0} E_1 \ar[r]^(.6){\odot_E}  \ar[d]_{L_1 \times_{L_0} L_1} \ar@{}[dr]|{\xRightarrow{\phi}}& E_1 \ar[d]^{L_1}\\
				B_1 \times_{B_0} E_1 \ar[r]_(.6){\odot_B} & B_1
		}}}
		\qquad \qquad \qquad
		\vcenter{\vbox{\xymatrix{
				E_0 \ar[d]_{L_0} \ar[r]^{i} \ar@{}[dr]|{\xRightarrow{\iota}} & E_1 \ar[d]^{L_1}\\
				B_0 \ar[r]_{i} & B_1	
		}}}
	\end{equation*}
	such that the source and target are strictly preserving. Moreover, $\phi$, $\iota$, and the associators and unitors of $\dcat{E}$ and $\dcat{B}$  satisfy the usual coherence conditions as in \cite{NMF}. A lax functor is called $\mathbf{strict}$ if $\phi$ and $\iota$ are identities.
\end{defn}	
	
\begin{remark}
	A strict functor in $\Cat$ corresponds to a strict functor between double categories.
\end{remark}

\begin{lemma}[\cite{Crutt}]
	Given a diagram $s \colon P_1 \to P_0 \leftarrow P_1 \colon t$ in $\mathbf{cFib}$, a pseudo category structure in $\mathbf{Fib}$ (resp. $\mathbf{cFib}$) is the same as two pseudo double category structure obtained by applying the top and bottom 2-functors,
\end{lemma}

\begin{defn}[\cite{Crutt}]
	A strict double functor $\dfib{P}{E}{B}$ between pseudo double categories given by 
	
	\begin{equation*}
		\xymatrix@=0.5cm{
			\cat{E}_1 \times_{\cat{E}_0} \cat{E}_1 \ar[rr]|-{\odot_\cat{E}} \ar[dd]_{\cat{F}_1 \times_{\cat{F}_0} \cat{F}_1} && \cat{E}_1 \ar[dd]_{\cat{F}_1} \ar@<1ex>[rr]^{s_\cat{E}} \ar@<-1ex>[rr]_{t_\cat{E}} && \cat{E}_0 \ar[dd]^{\cat{F}_0} \ar[ll]|-{i_\cat{E}} \\
			\\
			\cat{B}_1 \times_{\cat{B}_0} \cat{B}_1 \ar[rr]|-{\odot_\cat{B}} && \cat{B}_1 \ar@<1ex>[rr]^{s_\cat{B}} \ar@<-1ex>[rr]_{t_\cat{B}} && \cat{B}_0 \ar[ll]|-{i_\cat{B}}
		}
	\end{equation*}
	defines a \textbf{double fibration} when $P_0$ and $P_1$ are cloven fibrations, such that $s_{\cat{E}}$ and $t_{\cat{B}}$ preserve cleavages simultaneously, and $\odot_\cat{E}$ and $i_\cat{E}$ preserve cartesian morphisms.
\end{defn}
